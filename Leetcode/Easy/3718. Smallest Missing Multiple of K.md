[3718. Smallest Missing Multiple of K](https://leetcode.com/problems/smallest-missing-multiple-of-k/description/)

### Solution:

```class Solution {
public:
    int missingMultiple(vector<int>& nums, int k) {
        for(int i=0;i<nums.size();i++){
            int target = k*(i+1);

            // using the auto find to see if nums is present or not
            auto it = find(nums.begin(),nums.end(),target);
            // this is equal means the number is not found if found then not equal
            if(it == nums.end()){
                return target;
            }
        }
        // if already all present just return next multiple len(list)+1 * k
        return (nums.size()+1)*k;
    }
};```
